cmake_minimum_required(VERSION 3.16)
project(slam3d_mid360_dashboard LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(PCL_DIR "/usr/lib/x86_64-linux-gnu/cmake/pcl" CACHE PATH "ROS2 Humble PCL ABI" FORCE)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(MPI QUIET COMPONENTS C)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters io)
find_package(Boost REQUIRED COMPONENTS system)
add_executable(cloud_filter src/cloud_filter.cpp)
ament_target_dependencies(cloud_filter rclcpp sensor_msgs pcl_conversions)
target_include_directories(cloud_filter PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(cloud_filter ${PCL_LIBRARIES})
add_executable(map_preview src/map_preview.cpp)
ament_target_dependencies(map_preview rclcpp sensor_msgs std_srvs pcl_conversions)
target_include_directories(map_preview PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(map_preview ${PCL_LIBRARIES})
add_executable(web_dashboard src/web_dashboard.cpp)
ament_target_dependencies(web_dashboard rclcpp sensor_msgs nav_msgs diagnostic_msgs std_srvs std_msgs ament_index_cpp)
target_link_libraries(web_dashboard Boost::system pthread)
install(TARGETS cloud_filter map_preview web_dashboard DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config web DESTINATION share/${PROJECT_NAME})
ament_package()
