cmake_minimum_required(VERSION 3.16)
project(slam3d_relocalizer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
 set(CMAKE_BUILD_TYPE Release)
endif()
set(PCL_DIR "/usr/lib/x86_64-linux-gnu/cmake/pcl" CACHE PATH "ROS2 Humble PCL 1.12" FORCE)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(MPI QUIET COMPONENTS C)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters io registration kdtree)
add_executable(relocalizer_node src/relocalizer_node.cpp)
target_include_directories(relocalizer_node PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(relocalizer_node ${PCL_LIBRARIES})
ament_target_dependencies(relocalizer_node rclcpp nav_msgs sensor_msgs geometry_msgs
 std_msgs std_srvs diagnostic_msgs tf2_ros pcl_conversions)
add_executable(map_fixture src/map_fixture.cpp)
target_include_directories(map_fixture PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(map_fixture ${PCL_LIBRARIES})
ament_target_dependencies(map_fixture rclcpp nav_msgs sensor_msgs pcl_conversions)
install(TARGETS map_fixture DESTINATION lib/${PROJECT_NAME})
install(TARGETS relocalizer_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
