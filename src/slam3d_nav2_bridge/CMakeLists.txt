cmake_minimum_required(VERSION 3.16)
project(slam3d_nav2_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(PCL_DIR "/usr/lib/x86_64-linux-gnu/cmake/pcl" CACHE PATH "ROS2 Humble system PCL 1.12 ABI" FORCE)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(MPI QUIET COMPONENTS C)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
add_executable(odom_base_bridge src/odom_base_bridge.cpp)
ament_target_dependencies(odom_base_bridge rclcpp nav_msgs geometry_msgs tf2_ros)
target_link_libraries(odom_base_bridge Eigen3::Eigen)
add_executable(localization_stop_gate src/localization_stop_gate.cpp)
ament_target_dependencies(localization_stop_gate rclcpp nav_msgs geometry_msgs std_msgs
 std_srvs diagnostic_msgs tf2_ros tf2 rclcpp_action nav2_msgs)
add_executable(pcd_occupancy_map src/pcd_occupancy_map.cpp)
target_include_directories(pcd_occupancy_map PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(pcd_occupancy_map ${PCL_LIBRARIES})
ament_target_dependencies(pcd_occupancy_map rclcpp nav_msgs geometry_msgs pcl_conversions)
install(TARGETS odom_base_bridge localization_stop_gate pcd_occupancy_map
 DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config tools DESTINATION share/${PROJECT_NAME})
ament_package()
