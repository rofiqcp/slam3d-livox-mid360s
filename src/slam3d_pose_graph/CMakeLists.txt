cmake_minimum_required(VERSION 3.16)
project(slam3d_pose_graph LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(PCL_DIR "/usr/lib/x86_64-linux-gnu/cmake/pcl" CACHE PATH "ROS Humble PCL 1.12" FORCE)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(MPI QUIET COMPONENTS C)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters io registration)
find_package(GTSAM REQUIRED)
find_package(TBB REQUIRED)
add_executable(pose_graph_node src/pose_graph_node.cpp)
target_include_directories(pose_graph_node PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(pose_graph_node gtsam TBB::tbb ${PCL_LIBRARIES})
ament_target_dependencies(pose_graph_node rclcpp nav_msgs sensor_msgs geometry_msgs std_srvs std_msgs
  tf2_ros diagnostic_msgs pcl_conversions)
add_executable(loop_fixture src/loop_fixture.cpp)
ament_target_dependencies(loop_fixture rclcpp nav_msgs sensor_msgs)
install(TARGETS loop_fixture DESTINATION lib/${PROJECT_NAME})
install(TARGETS pose_graph_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
